💻 Implementation Code

/* CSS Grid Gallery */ .gallery { display: grid; grid-template-columns: repeat(auto-fit, minmax(350px, 1fr)); gap: 30px; } /* Gallery Item with Hover Effect */ .gallery-item { border-radius: 15px; overflow: hidden; transition: transform 0.4s; } .gallery-item:hover { transform: translateY(-10px); } /* HTML Structure */ <div class="gallery"> <div class="gallery-item"> <figure> <img src="image.jpg" alt="Description"> <figcaption> <h3>Image Title</h3> <p>Image description</p> </figcaption> </figure> </div> </div>

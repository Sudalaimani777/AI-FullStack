1. WHAT IS IIFE?
================

Definition:
-----------
IIFE stands for "Immediately Invoked Function Expression"
It's a function that runs as soon as it's defined.

Breaking Down the Acronym:
--------------------------
- Immediately = Executes right away
- Invoked = Called/Executed
- Function = Block of reusable code
- Expression = Code that produces a value

Simple Analogy:
---------------
Think of IIFE like a self-destructing message:
- You write it
- It reads itself immediately
- Then it's gone (contents stay private)


2. SYNTAX & STRUCTURE
=====================

A. BASIC SYNTAX
---------------
(function() {
    // Code here runs immediately
    console.log("I run right away!");
})();

Breaking it down:
Step 1: (function() { ... })   ← Wrap function in parentheses (makes it an expression)
Step 2: ()                     ← Call it immediately


B. TWO VALID SYNTAXES
----------------------
// Style 1: Parentheses wrap everything
(function() {
    console.log("Style 1");
})();

// Style 2: Parentheses wrap only function
(function() {
    console.log("Style 2");
}());

Both work identically! Style 1 is more common.


C. ARROW FUNCTION IIFE (ES6)
-----------------------------
(() => {
    console.log("Modern IIFE!");
})();

// One-liner
(() => console.log("Short IIFE"))();


D. NAMED vs ANONYMOUS IIFE
---------------------------
// Anonymous (most common)
(function() {
    console.log("No name");
})();

// Named (useful for debugging/recursion)
(function myIIFE() {
    console.log("I have a name");
})();


3. WHY USE IIFE?
================

A. ENCAPSULATION & PRIVACY
---------------------------
Problem: Variables pollute global scope
Solution: IIFE creates private scope

// Without IIFE - Global pollution
var password = "secret123";
var apiKey = "xyz789";
// Anyone can access these! ❌

// With IIFE - Private scope
(function() {
    var password = "secret123";
    var apiKey = "xyz789";
    // These are private! ✅
})();

console.log(password); // undefined - protected!


B. AVOID GLOBAL SCOPE POLLUTION
--------------------------------
Prevents naming conflicts:

// File 1
var counter = 0;

// File 2 (accidentally overrides!)
var counter = 100;  // Oops! Overwrote the first one ❌

// Better approach with IIFE
(function() {
    var counter = 0;
    // Safe inside here ✅
})();

(function() {
    var counter = 100;
    // Also safe, doesn't conflict ✅
})();


C. CREATE PRIVATE VARIABLES
----------------------------
(function() {
    var secret = "Can't access me from outside!";
    
    function privateFunction() {
        console.log(secret);
    }
    
    privateFunction(); // Works inside
})();

console.log(secret); // Error: secret is not defined ✅


D. ONE-TIME INITIALIZATION
---------------------------
Run setup code that only needs to execute once:

(function() {
    console.log("App initializing...");
    // Set up event listeners
    // Configure settings
    // Load initial data
    console.log("App ready!");
})();


4. IIFE PATTERNS
================

A. BASIC PATTERN
----------------
(function() {
    console.log("Basic IIFE");
})();


B. WITH PARAMETERS
------------------
(function(name, age) {
    console.log(`${name} is ${age} years old`);
})("Alice", 25);

// Passing global objects
(function(window, document) {
    // Now can use window and document
    console.log(window.innerWidth);
})(window, document);


C. RETURNING VALUES
-------------------
var result = (function() {
    return "I can return values!";
})();

console.log(result); // "I can return values!"

// Return object
var calculator = (function() {
    return {
        add: function(a, b) { return a + b; },
        subtract: function(a, b) { return a - b; }
    };
})();

console.log(calculator.add(5, 3)); // 8


D. MODULE PATTERN
-----------------
var myModule = (function() {
    // Private variables
    var privateVar = "I'm private";
    var privateCounter = 0;
    
    // Private function
    function privateFunction() {
        console.log(privateVar);
    }
    
    // Public API (returned object)
    return {
        publicMethod: function() {
            privateCounter++;
            privateFunction();
        },
        getCount: function() {
            return privateCounter;
        }
    };
})();

myModule.publicMethod(); // Works
console.log(myModule.getCount()); // 1
console.log(myModule.privateVar); // undefined - private!


E. REVEALING MODULE PATTERN
----------------------------
var myModule = (function() {
    var privateVar = "secret";
    
    function privateFunction() {
        console.log(privateVar);
    }
    
    function publicFunction() {
        privateFunction();
    }
    
    function anotherPublic() {
        console.log("Another public method");
    }
    
    // Reveal only what you want
    return {
        doSomething: publicFunction,
        doAnother: anotherPublic
    };
})();

myModule.doSomething();


5. PRACTICAL EXAMPLES
======================

EXAMPLE 1: Counter with Private State
--------------------------------------
var counter = (function() {
    var count = 0; // Private variable
    
    return {
        increment: function() {
            count++;
            console.log(count);
        },
        decrement: function() {
            count--;
            console.log(count);
        },
        getCount: function() {
            return count;
        },
        reset: function() {
            count = 0;
        }
    };
})();

counter.increment(); // 1
counter.increment(); // 2
counter.decrement(); // 1
console.log(counter.count); // undefined - private!
console.log(counter.getCount()); // 1 - only through method


EXAMPLE 2: Configuration Manager
---------------------------------
var config = (function() {
    var settings = {
        apiUrl: "https://api.example.com",
        timeout: 5000,
        debug: false
    };
    
    return {
        get: function(key) {
            return settings[key];
        },
        set: function(key, value) {
            if (settings.hasOwnProperty(key)) {
                settings[key] = value;
            }
        },
        getAll: function() {
            return Object.assign({}, settings); // Return copy
        }
    };
})();

console.log(config.get('apiUrl')); // "https://api.example.com"
config.set('timeout', 10000);
console.log(config.get('timeout')); // 10000


EXAMPLE 3: jQuery-like Library Pattern
---------------------------------------
var $ = (function() {
    function querySelector(selector) {
        return document.querySelector(selector);
    }
    
    function addEvent(selector, event, handler) {
        var element = querySelector(selector);
        if (element) {
            element.addEventListener(event, handler);
        }
    }
    
    return {
        select: querySelector,
        on: addEvent
    };
})();

// Usage
$.on('#myButton', 'click', function() {
    console.log('Button clicked!');
});


EXAMPLE 4: Namespace Creation
------------------------------
var APP = APP || {};

APP.utils = (function() {
    function formatDate(date) {
        return date.toLocaleDateString();
    }
    
    function capitalize(str) {
        return str.charAt(0).toUpperCase() + str.slice(1);
    }
    
    return {
        formatDate: formatDate,
        capitalize: capitalize
    };
})();

APP.auth = (function() {
    var isLoggedIn = false;
    
    return {
        login: function() {
            isLoggedIn = true;
        },
        logout: function() {
            isLoggedIn = false;
        },
        checkAuth: function() {
            return isLoggedIn;
        }
    };
})();

// Usage
APP.utils.capitalize("hello"); // "Hello"
APP.auth.login();


EXAMPLE 5: One-Time Initialization
-----------------------------------
(function() {
    // Run only once when script loads
    console.log("Initializing app...");
    
    // Set up event listeners
    document.addEventListener('DOMContentLoaded', function() {
        console.log("DOM ready!");
    });
    
    // Configure defaults
    var config = {
        theme: 'dark',
        language: 'en'
    };
    
    // Apply theme
    document.body.classList.add(config.theme + '-theme');
    
    console.log("App initialized!");
})();


EXAMPLE 6: Loop Variable Problem (Classic)
-------------------------------------------
// Problem: All buttons alert "5"
for (var i = 0; i < 5; i++) {
    setTimeout(function() {
        console.log(i); // All print 5!
    }, 1000);
}

// Solution 1: IIFE
for (var i = 0; i < 5; i++) {
    (function(j) {
        setTimeout(function() {
            console.log(j); // Prints 0, 1, 2, 3, 4
        }, 1000);
    })(i);
}

// Solution 2: let (modern)
for (let i = 0; i < 5; i++) {
    setTimeout(function() {
        console.log(i); // Prints 0, 1, 2, 3, 4
    }, 1000);
}


6. ADVANCED TECHNIQUES
======================

A. PASSING GLOBAL OBJECTS
--------------------------
Why? Faster lookups and safer references

(function(window, document, undefined) {
    // window, document are now local parameters
    // undefined is truly undefined (can't be overridden)
    
    console.log(window.innerWidth);
    console.log(document.title);
    
})(window, document);

Benefits:
- Faster: Local variables are faster to access
- Minification: Parameters can be minified
- Safety: Protected from global changes


B. CONDITIONAL IIFE
--------------------
var result = condition ? 
    (function() {
        return "Option A";
    })() : 
    (function() {
        return "Option B";
    })();


C. RECURSIVE IIFE
------------------
(function countdown(n) {
    console.log(n);
    if (n > 0) {
        countdown(n - 1);
    }
})(5);
// Prints: 5, 4, 3, 2, 1, 0


D. AUGMENTATION PATTERN
------------------------
Extend existing modules:

var MODULE = (function(mod) {
    mod.newMethod = function() {
        console.log("New method added!");
    };
    return mod;
})(MODULE || {});


E. CHAINING IIFE
-----------------
var app = (function() {
    return {
        init: function() {
            console.log("Initialized");
            return this;
        },
        start: function() {
            console.log("Started");
            return this;
        }
    };
})();

app.init().start(); // Chaining


7. COMMON USE CASES
===================

✅ 1. Creating Modules/Libraries
    var MyLibrary = (function() { ... })();

✅ 2. Avoiding Global Variables
    (function() { var temp = 123; })();

✅ 3. Protecting Code from Others
    (function() { /* Your code */ })();

✅ 4. Creating Private Variables/Functions
    var module = (function() { var private = 1; })();

✅ 5. One-Time Initialization
    (function() { setupApp(); })();

✅ 6. Aliasing Global Objects
    (function($) { /* Use jQuery as $ */ })(jQuery);

✅ 7. Creating Namespaces
    var APP = APP || {};
    APP.module = (function() { ... })();

✅ 8. Fixing Variable Scope Issues in Loops
    for (var i = 0; i < 5; i++) {
        (function(j) { ... })(i);
    }

✅ 9. Testing Code in Isolation
    (function() { /* Test code */ })();

✅ 10. Plugin Development
    (function($, window, document) { ... })(jQuery, window, document);


8. BEST PRACTICES
=================

✅ DO:
------
1. Use IIFE for Module Pattern
   var module = (function() { return {}; })();

2. Pass Global Objects as Parameters
   (function(window, $) { })(window, jQuery);

3. Return Public API from IIFE
   return { publicMethod: function() {} };

4. Use Semicolon Before IIFE
   ;(function() { })();
   // Prevents errors if previous line missing semicolon

5. Use Strict Mode
   (function() {
       'use strict';
       // Your code
   })();

6. Name IIFE for Better Debugging
   (function myModule() { ... })();

7. Keep IIFE Focused (Single Responsibility)
   One IIFE = One purpose


❌ DON'T:
---------
1. Overuse IIFE in Modern Code
   // ES6 modules are usually better
   
2. Create Too Many Global Variables
   var a = (function(){})();
   var b = (function(){})();
   // Consider one namespace instead

3. Forget Parentheses
   function() { }();  // ❌ Syntax Error
   (function() { })(); // ✅ Correct

4. Make IIFE Too Large
   // Keep it manageable, split if needed

5. Use IIFE When Simple Block Scope is Enough
   // Don't need IIFE:
   {
       let temp = 123;
   }


9. COMMON MISTAKES
==================

MISTAKE 1: Missing Parentheses
-------------------------------
// ❌ Wrong
function() {
    console.log("Error!");
}();

// ✅ Correct
(function() {
    console.log("Works!");
})();


MISTAKE 2: Trying to Call Named Function
-----------------------------------------
// ❌ Wrong
(function myFunc() {
    console.log("Inside");
})();

myFunc(); // Error: myFunc is not defined

// The name is only available INSIDE the function
(function myFunc() {
    console.log("I can call myself");
    // myFunc(); would work here for recursion
})();


MISTAKE 3: Forgetting Return Statement
---------------------------------------
// ❌ Wrong - returns undefined
var module = (function() {
    var x = 1;
    // Forgot to return!
})();

console.log(module); // undefined

// ✅ Correct
var module = (function() {
    var x = 1;
    return { getX: function() { return x; } };
})();


MISTAKE 4: Variable Scope Issues
---------------------------------
// ❌ Problem
for (var i = 0; i < 3; i++) {
    setTimeout(function() {
        console.log(i); // All print 3
    }, 100);
}

// ✅ Solution
for (var i = 0; i < 3; i++) {
    (function(j) {
        setTimeout(function() {
            console.log(j); // Prints 0, 1, 2
        }, 100);
    })(i);
}


MISTAKE 5: Not Using Semicolon Before IIFE
-------------------------------------------
// ❌ Can cause issues
var x = 5
(function() { })()
// JavaScript thinks: var x = 5(function() { })()

// ✅ Safe
var x = 5;
(function() { })();

// ✅ Or use leading semicolon
var x = 5
;(function() { })();


10. MODERN ALTERNATIVES
========================

A. ES6 BLOCK SCOPE (let/const)
-------------------------------
// Old way (IIFE)
(function() {
    var temp = "private";
    console.log(temp);
})();

// Modern way (Block scope)
{
    let temp = "private";
    console.log(temp);
}


B. ES6 MODULES
--------------
// Old way (IIFE)
var myModule = (function() {
    var private = 1;
    return {
        public: function() { }
    };
})();

// Modern way (ES6 Module)
// module.js
let private = 1;
export function public() { }


C. ARROW FUNCTIONS
------------------
// Old way
(function() {
    console.log("Old");
})();

// Modern way
(() => {
    console.log("Modern");
})();


D. ASYNC IIFE
--------------
// For async/await operations
(async () => {
    const data = await fetch('https://api.example.com');
    console.log(data);
})();


WHEN TO USE WHICH:
------------------
Use IIFE when:
- Need immediate execution
- Creating module pattern
- Supporting older browsers
- Need private scope quickly

Use ES6 Modules when:
- Building modern applications
- Need import/export
- Want better tooling support
- Working with bundlers

Use Block Scope when:
- Just need temporary variable
- Modern browser only
- Simple scope isolation
